import streamlit as st
from serpapi import GoogleSearch
from bs4 import BeautifulSoup
import requests
from openai import OpenAI
from wordpress_xmlrpc import Client, WordPressPost
from wordpress_xmlrpc.methods.posts import NewPost

# 앱 타이틀
st.set_page_config(page_title="수익형 SEO 포스팅 봇", layout="wide")
st.title("💰 AI 수익형 포스팅 자동화 시스템")

# 1. 사이드바 설정 (정보 입력)
with st.sidebar:
    st.header("⚙️ 필수 설정")
    wp_url = st.text_input("워드프레스 주소 (xmlrpc.php 포함)", placeholder="https://example.com/xmlrpc.php")
    wp_user = st.text_input("워드프레스 아이디")
    wp_pw = st.text_input("워드프레스 앱 비밀번호", type="password")
    openai_key = st.text_input("OpenAI API 키", type="password")
    serp_key = st.text_input("SerpApi 키", type="password")
    st.info("💡 모든 정보를 입력해야 버튼이 활성화됩니다.")

# 2. 분석 및 본문 수집 함수
def get_competitor_analysis(keyword, api_key):
    params = {"engine": "google", "q": keyword, "api_key": api_key, "hl": "ko", "gl": "kr"}
    search = GoogleSearch(params)
    results = search.get_dict()
    organic = results.get("organic_results", [])
    
    st.subheader("🔍 검색 결과 분석 현황")
    comp_content = ""
    target_link = ""
    
    for res in organic[:5]:
        link = res.get("link")
        domain = link.split('/')[2]
        # 개인이 공략 가능한 도메인 필터링
        is_portable = any(x in domain for x in ['tistory.com', 'm.blog.naver.com', 'wordpress.com', 'tistory.io'])
        
        icon = "✅" if is_portable else "❌"
        st.write(f"{icon} {domain} | {res.get('title')[:30]}...")
        
        # 첫 번째 공략 가능한 블로그의 글을 분석 대상으로 삼음
        if is_portable and not comp_content:
            target_link = link
            try:
                resp = requests.get(link, headers={'User-Agent': 'Mozilla/5.0'}, timeout=7)
                soup = BeautifulSoup(resp.text, 'html.parser')
                comp_content = " ".join([p.get_text() for p in soup.find_all(['p', 'h2', 'h3'])])[:2500]
            except:
                comp_content = res.get("snippet")
                
    return comp_content, target_link

# 3. 메인 실행부
keyword = st.text_input("공략하고 싶은 키워드를 입력하세요", placeholder="예: 정부지원금 신청방법")

if st.button("🚀 분석 및 포스팅 시작"):
    if not (wp_url and wp_user and wp_pw and openai_key and serp_key):
        st.error("⚠️ 사이드바에서 모든 설정값을 입력해주세요.")
    else:
        # STEP 1: 경쟁사 및 영역 분석
        with st.spinner("구글 검색 결과 및 경쟁사 글을 정밀 분석 중..."):
            comp_text, ref_url = get_competitor_analysis(keyword, serp_key)
            
        # STEP 2: AI 글쓰기 (비교 우위 + 수익형 버튼 전략)
        with st.spinner("AI가 더 나은 글을 작성하며 수익형 버튼을 삽입 중입니다..."):
            client = OpenAI(api_key=openai_key)
            
            prompt = f"""
            너는 연봉 10억의 SEO 전략가이자 전문 작가야. 
            키워드 '{keyword}'에 대해 경쟁사보다 검색 순위가 더 높게 나오도록 글을 써줘.
            
            [분석된 경쟁사 내용]
            {comp_text}
            
            [작성 지침 - 반드시 지킬 것]
            1. 경쟁사보다 1.5배 더 풍부한 정보를 담을 것.
            2. 본문 시작 직후에 3줄 핵심 요약을 넣을 것.
            3. 본문 중간(H2 소제목 아래)과 글 제일 마지막에 클릭을 유도하는 '수익형 버튼'을 넣을 것.
            4. 버튼은 반드시 아래의 HTML 코드를 사용하고 링크(href)는 '#' 또는 관련 정보 주소로 채울 것:
               <div style="text-align: center; margin: 30px 0;">
                 <a href="#" style="background-color: #ef4444; color: white; padding: 18px 30px; text-decoration: none; border-radius: 50px; font-weight: bold; font-size: 1.2rem; display: inline-block; box-shadow: 0 10px 15px rgba(0,0,0,0.1);">👉 지금 바로 확인하기 (클릭)</a>
               </div>
            5. 워드프레스에 바로 적용되도록 마크다운(H2, H3) 형식을 철저히 지킬 것.
            """
            
            response = client.chat.completions.create(
                model="gpt-4o",
                messages=[{"role": "user", "content": prompt}]
            )
            final_text = response.choices[0].message.content
            title = final_text.split('\n')[0].replace('#', '').strip()

        # STEP 3: 워드프레스 포스팅
        with st.spinner("워드프레스로 전송 중..."):
            try:
                wp = Client(wp_url, wp_user, wp_pw)
                post = WordPressPost()
                post.title = title
                post.content = final_text
                post.post_status = 'publish'
                post.terms_names = {'category': ['수익화 정보']}
                wp.call(NewPost(post))
                
                st.success(f"✅ 포스팅 완료! 제목: {title}")
                st.balloons()
                if ref_url:
                    st.write(f"🔗 참고한 경쟁사 주소: {ref_url}")
            except Exception as e:
                st.error(f"❌ 포스팅 중 오류 발생: {e}")
